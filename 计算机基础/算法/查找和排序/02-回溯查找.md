# 回溯查找
> 回溯查找实际是一个暴力枚举的过程，最坏的情况，需要遍历所有的元素，但如果使用回溯的办法，那么就意味没有更优的解法

## 题目一
N皇后

## 解法
> 因为回溯查找不能够重复查找已经找过的路径，所以需要一个记忆矩阵，用于记录已经走过的路

```Swift
class Solution {
    var res:[[String]] = []
    func solveNQueens(_ n: Int) -> [[String]] {
        var line = String(Array(repeating:".", count:n))
        var board:[String] = Array(repeating:line, count:n)
        backtrace(0, &board)
        return res
    }

    func backtrace(_ row:Int, _ board: inout [String]) {
        if row == board.count {
            res.append(board)
            return
        }
        var n = board[0].count
        for col in 0...n-1 {
            if !isValid(board, row, col) {
                continue
            }

            var line: String = board[row]
            var chars: [Character] = Array(line)
            chars[col] = "Q"
            board[row] = String(chars)
            backtrace(row + 1, &board)
            line = board[row]
            chars = Array(line)
            chars[col] = "."
            board[row] = String(chars)
        }

    }

    func isValid(_ board:[String], _ row: Int, _ col: Int) -> Bool{
        var n:Int = board.count
        // 列检查
        var i  = 0
        while i < row {
            var chars:[Character] = Array(board[i])
            if chars[col] == "Q" {
                return false
            }
            i += 1
        }
        // 右上方检查
        var a = row - 1, b = col + 1
        while a >= 0 && b < n {
            var chars:[Character] = Array(board[a])
            if chars[b] == "Q" {
                return false
            }
            a -= 1
            b += 1
        }

        // 左上方检查
        var c = row - 1, d = col - 1
        while c >= 0 && d >= 0 {
            var chars:[Character] = Array(board[c])
            if chars[d] == "Q" {
                return false
            }
            c -= 1
            d -= 1
        }


        return true
    }
}
```
