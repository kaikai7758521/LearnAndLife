# 倒序打印链表
## 题目
> 先有一个链表，请倒序打印链表的每个节点的值
```Swift
class ListNode {
    var value: Int
    var next: ListNode?
    // 省略初始化方法    
}
```

## 解法一：递归
我们可以直接使用递归的办法，实现链表的倒序打印
```Swift
func printListNode(_ head:ListNode) {
    var node = head
    printListNode(node)
    print(node.value)
}
```

但是在实际过程中，链表的长度过长将会导致栈溢出！！

## 解法二：反转链表
我们可以直接将链表反转，在反转完毕后，依次打印，这样就不存在栈溢出的风险了

```Swift
func reverseListNode(_ head: ListNode?) -> ListNode? {
    var newHead: ListNode? = nil
    var node: ListNode? = head
    var preNode: ListNode? = nil
    while node != nil {
        let tempNode = node?.next
        node?.next = preNode
        preNode = node
        node = tempNode
    }
    return node
}

func printListNode(_ head: ListNode?) {
    var node: ListNode? = head
    while node != nil {
        print(node?.value)
        node = node?.next    
    }
}
```
