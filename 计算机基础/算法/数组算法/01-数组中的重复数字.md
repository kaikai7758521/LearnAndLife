# 数组中重复的数字
## 题目一
有一个数长度为n的数组，数组里所有的数字都在0～n-1之间，其中有某些数字是重复的，找出任意一个重复数字

## 解法
暴力解法：使用2层for循环，在原有的空间上不开辟新的空间，直接可以找到

使用Set辅助：一次for循环，每次元素放入set中，直到判断出当前元素已经在set中重复

修改数组实现：尝试将每个元素放到其正确的位置
```Swift
func findDuplicateNumber(_ nums:[Int]) -> Int {
    var newNums: [Int] = nums // 由于swift不支持修改参数，所以必须复制一个
    for i in 0...newNums.count-1 {
        while newNums[i] != i {
            if newNums[i] == newNums[newNums[i]] {
                return newNums[i]
            } else {
                var temp = newNums[i]
                newNums[i] = newNums[temp]
                newNums[temp] = temp
            }
        }
    }
    return -1
}
```


## 题目二
有一个数长度为n+1的数组，数组里所有的数字都在1～n之间，其中有某些数字是重复的，找出任意一个（要求修改数组）
这里参考二分法进行查找
思路是：按照数组的长度进行切分，如果某个片段的数字长度大于该片段本身的数字，那么就存在重复，我们对这段片段进行多次二分查找，直到片段切分为1，此时如果切分的数组长度仍然大于1，那么此时片段中就全部为重复的数字，范围第一个即可
```Swift
func findDuplicateNumberAndNotModify(_ nums:[Int], _ length:Int) -> Int {
    if nums.count <= 0 {
        return -1
    }
    
    var start = 1
    var end = length - 1
    while end >= start {
        var middle = (end - start) / 2 + start
        var countRange = countRange(nums, length, start, middle)
        if end == start {
            if countRange > 1 {
                return start
            } else {
                break
            }
        }
        if countRange > middle - start + 1 {
            end = middle
        } else {
            start = middle + 1
        }
    }
    
    func countRange(_ nums:[Int], _ length:Int, _ start: Int, _ end: Int) -> Int {
        var countRange = 0
        if nums.count <= 0 {
            return 0
        }
        for i in 0...length-1 {
            if nums[i] >= start && nums[i] <= end {
                countRange += 1
            }
        }
        return countRange
    }
    
    return -1
}
```
