# Print linked list in reverse order
## Question
> There is a linked list first. Please print the value of each node of the linked list in reverse order.
```Swift
class ListNode {
    var value: Int
    var next: ListNode?
    // 省略初始化方法    
}
```

## Solution 1: Recursion
We can directly use recursion to print the linked list in reverse order.
```Swift
func printListNode(_ head:ListNode) {
    var node = head
    printListNode(node)
    print(node.value)
}
```

But in the actual process, if the length of the linked list is too long, it will cause stack overflow! !

## Solution 2: Reverse the linked list
We can directly reverse the linked list, and after the reversal is completed, print in sequence, so there is no risk of stack overflow.

```Swift
func reverseListNode(_ head: ListNode?) -> ListNode? {
    var newHead: ListNode? = nil
    var node: ListNode? = head
    var preNode: ListNode? = nil
    while node != nil {
        let tempNode = node?.next
        node?.next = preNode
        preNode = node
        node = tempNode
    }
    return node
}

func printListNode(_ head: ListNode?) {
    var node: ListNode? = head
    while node != nil {
        print(node?.value)
        node = node?.next    
    }
}
```
