# Backtracking search
> Backtracking search is actually a violent enumeration process. In the worst case, all elements need to be traversed, but if the backtracking method is used, it means that there is no better solution.

## Question 1
Queen N

## Solution
> Because backtracking search cannot repeatedly search the path that has already been found, a memory matrix is needed to record the path that has been traveled.

```Swift
class Solution {
    var res:[[String]] = []
    func solveNQueens(_ n: Int) -> [[String]] {
        var line = String(Array(repeating:".", count:n))
        var board:[String] = Array(repeating:line, count:n)
        backtrace(0, &board)
        return res
    }

    func backtrace(_ row:Int, _ board: inout [String]) {
        if row == board.count {
            res.append(board)
            return
        }
        var n = board[0].count
        for col in 0...n-1 {
            if !isValid(board, row, col) {
                continue
            }

            var line: String = board[row]
            var chars: [Character] = Array(line)
            chars[col] = "Q"
            board[row] = String(chars)
            backtrace(row + 1, &board)
            line = board[row]
            chars = Array(line)
            chars[col] = "."
            board[row] = String(chars)
        }

    }

    func isValid(_ board:[String], _ row: Int, _ col: Int) -> Bool{
        var n:Int = board.count
        // 列检查
        var i  = 0
        while i < row {
            var chars:[Character] = Array(board[i])
            if chars[col] == "Q" {
                return false
            }
            i += 1
        }
        // 右上方检查
        var a = row - 1, b = col + 1
        while a >= 0 && b < n {
            var chars:[Character] = Array(board[a])
            if chars[b] == "Q" {
                return false
            }
            a -= 1
            b += 1
        }

        // 左上方检查
        var c = row - 1, d = col - 1
        while c >= 0 && d >= 0 {
            var chars:[Character] = Array(board[c])
            if chars[d] == "Q" {
                return false
            }
            c -= 1
            d -= 1
        }


        return true
    }
}
```
