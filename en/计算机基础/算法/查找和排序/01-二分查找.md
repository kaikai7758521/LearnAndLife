# Binary search
## Question
Given an array, this array is a rotated array, that is, the first few numbers are placed at the end of the array, and you need to find the minimum value of the current array.

## Solution
This is a typical binary search problem:
We use binary search to find the inverted slice, and then perform binary search within the current slice until we find the minimum value.

```Swift
func findMinNumberInNums(_ nums: [Int]) -> Int {
    var pre = 0, after = nums.count - 1
    var mid = pre
    
    // [5, 1, 2, 3, 4, 4]
    while nums[pre] >= nums[after] {
        if (after - pre) == 1 {
            mid = after
            break
        }
        mid = (after - pre) / 2 + pre
        
        if nums[pre] == nums[after] && nums[mid] == nums[pre]{
            var min = nums[0]
            for i in nums {
                if min >= i {
                    min = i
                }
                return min
            }
        }
        
        if nums[mid] <= nums[pre] {
            after = mid
        } else if nums[mid] >= nums[pre] {
            pre = mid
        }
        
    }
    
    return nums[mid]
    }
```
